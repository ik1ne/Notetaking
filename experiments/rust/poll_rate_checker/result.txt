#!/usr/bin/env python3
import sys
import re
import statistics

# match lines like:
# PointerID 182: Δ = 1.1224ms, Polling rate ≈ 890.95 Hz
rate_re = re.compile(r'Polling rate ≈\s*([\d.]+)\s*Hz')

def main():
    rates = []
    for line in sys.stdin:
        m = rate_re.search(line)
        if m:
            rates.append(float(m.group(1)))

    if not rates:
        print("No samples found on stdin", file=sys.stderr)
        sys.exit(1)

    print(f"Samples: {len(rates)}")
    print(f"Average rate: {statistics.mean(rates):.2f} Hz")
    print(f"Min rate:     {min(rates):.2f} Hz")
    print(f"Max rate:     {max(rates):.2f} Hz")
    print(f"Median rate:  {statistics.median(rates):.2f} Hz")
    # only compute stdev if at least 2 samples
    if len(rates) > 1:
        print(f"Std dev:      {statistics.stdev(rates):.2f} Hz")

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
import sys
import re

# match lines like:
# PointerID 182: Δ = 1.1224ms, Polling rate ≈ 890.95 Hz
latency_re = re.compile(r'Δ\s*=\s*([\d.]+)\s*ms')

# hardcoded threshold in milliseconds
THRESHOLD_MS = 10.0

def main():
    latencies = []
    for line in sys.stdin:
        m = latency_re.search(line)
        if m:
            latencies.append(float(m.group(1)))

    if not latencies:
        print("No latency samples found on stdin", file=sys.stderr)
        sys.exit(1)

    total = len(latencies)
    above = sum(1 for l in latencies if l > THRESHOLD_MS)
    percent = above / total * 100

    print(f"Threshold: {THRESHOLD_MS} ms")
    print(f"Samples above threshold: {above} / {total} ({percent:.2f}%)")

if __name__ == "__main__":
    main()
